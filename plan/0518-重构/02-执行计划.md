### 1.4 TUI 事件循环集成

**文件**：`crates/cowd-cli/src/main.rs`（修改 `run_tui_repl`）

```rust
fn run_tui_repl(mut cli: LiveCli, workspace: PathBuf) -> Result<()> {
    let (tui_tx, tui_rx) = mpsc::channel::<TuiEvent>(256);
    let mut app = tui::App::new(...);
    let mut turn_handle: Option<thread::JoinHandle<TurnOutcome>> = None;

    loop {
        // 1. 收集事件
        drain_tui_events(&tui_rx, &mut app);

        // 2. 检查 turn 完成
        if let Some(handle) = &turn_handle {
            if handle.is_finished() {
                match handle.join() {
                    Ok(TurnOutcome::Success { session, summary }) => {
                        cli.runtime.update_session(session);
                        cli.persist_session()?;
                        app.finish_turn(summary);
                    }
                    Ok(TurnOutcome::Error(e)) => {
                        app.show_error(&e.to_string());
                    }
                    _ => {}
                }
                turn_handle = None;
            }
        }

        // 3. 处理输入
        match tui::input::handle_input(&mut app)? {
            InputResult::Submit(text) if turn_handle.is_none() => {
                let session = cli.runtime.session().without_persistence();
                turn_handle = Some(spawn_turn(text, session, cli.model.clone(), tui_tx.clone()));
            }
            InputResult::Exit => break,
            _ => {}
        }

        // 4. 渲染
        terminal.draw(|f| tui::render::draw(f, &mut app))?;

        // 5. 帧率控制
        thread::sleep(Duration::from_millis(8)); // ~120fps max
    }
}
```

**验收**：
- [ ] TUI 不阻塞，输入响应 <50ms
- [ ] 流式文本逐字符渲染
- [ ] 多个 turn 可串行执行
- [ ] Ctrl+C 中断及时

```rust
pub fn draw(frame: &mut Frame, app: &mut App) {
    let area = frame.area();

    // 垂直布局：状态栏(1) | 主区域(fill) | 输入栏(3)
    let vchunks = Layout::vertical([
        Constraint::Length(1),
        Constraint::Fill(1),
        Constraint::Length(3),
    ]).split(area);

    // 状态栏
    draw_status_bar(vchunks[0], app, frame);

    // 主面板（按 panel 类型切换）
    match app.current_panel {
        Panel::Chat | Panel::Gateway | Panel::Delegate => {
            // 聊天区域布局：消息(70%) | 思考面板(30%, 可折叠)
            let (chat_area, thinking_area) = if !app.streaming_thinking.is_empty() {
                let chunks = Layout::vertical([
                    Constraint::Fill(1),
                    Constraint::Length(app.thinking_height()),
                ]).split(vchunks[1]);
                (chunks[0], Some(chunks[1]))
            } else {
                (vchunks[1], None)
            };

            draw_chat_messages(chat_area, app, frame);

            if let Some(area) = thinking_area {
                draw_thinking_panel(area, app, frame);
            }

            // 覆盖面板
            if app.current_panel == Panel::Gateway {
                draw_gateway_panel(area, app, frame);
            }
            if app.current_panel == Panel::Delegate {
                draw_delegate_panel(area, app, frame);
            }
        }
        Panel::Files => draw_file_browser(vchunks[1], app, frame),
        Panel::Memory => draw_memory_panel(vchunks[1], app, frame),
        Panel::Skills => draw_skills_panel(vchunks[1], app, frame),
    }

    // 输入栏
    draw_input_bar(vchunks[2], app, frame);
}
```

```rust
// 启动时从 session.messages 加载到 app.messages
fn load_session_messages(session: &Session, app: &mut App) {
    for msg in &session.messages {
        match msg.role {
            MessageRole::User => {
                for block in &msg.blocks {
                    if let ContentBlock::Text { text } = block {
                        app.add_message("user", text);
                    }
                }
            }
            MessageRole::Assistant => {
                let mut text = String::new();
                for block in &msg.blocks {
                    match block {
                        ContentBlock::Text { text: t } => text.push_str(t),
                        ContentBlock::Thinking { .. } => {}, // 不显示历史 thinking
                        ContentBlock::ToolUse { name, .. } => {
                            app.live_tool_cards.push(LiveToolCard {
                                name: name.clone(),
                                status: ToolStatus::Done,
                                ..
                            });
                        }
                        _ => {}
                    }
                }
                if !text.is_empty() {
                    app.add_message("assistant", &text);
                }
            }
            _ => {}
        }
    }
}
```

### 4.1 事件批处理

```rust
fn drain_tui_events(rx: &mpsc::Receiver<TuiEvent>, app: &mut App) -> usize {
    let mut count = 0;
    while let Ok(event) = rx.try_recv() {
        app.apply_event(event);
        count += 1;
        if count >= 64 { break; } // 每帧最多处理 64 个事件
    }
    count
}
```
