fn export_migration_snapshot(
    connection: &impl crate::database::ConnectionLike,
) -> Result<MfgMigrationSnapshot, MfgRepositoryError> {
    let schema_version = schema_version(connection)?;
    let mut tables = BTreeMap::new();
    for table_name in MFG_MIGRATION_TABLES {
        let columns = migration_columns(connection, table_name)?;
        let column_list = columns
            .iter()
            .map(|column| quote_migration_identifier(&column.name))
            .collect::<Result<Vec<_>, _>>()?
            .join(", ");
        let rows = connection
            .query_rows(&format!(
                "SELECT {column_list} FROM {}",
                quote_migration_identifier(table_name)?
            ))?
            .into_iter()
            .map(|row| {
                if row.values().len() != columns.len() {
                    return Err(MfgRepositoryError::MigrationRejected(format!(
                        "table {table_name} returned {} values for {} columns",
                        row.values().len(),
                        columns.len()
                    )));
                }
                Ok(row
                    .values()
                    .iter()
                    .map(migration_value_from_database)
                    .collect())
            })
            .collect::<Result<Vec<Vec<_>>, _>>()?;
        let mut rows = rows;
        rows.sort_by(|left, right| {
            serde_json::to_vec(left)
                .expect("migration values serialize")
                .cmp(&serde_json::to_vec(right).expect("migration values serialize"))
        });
        tables.insert(table_name.to_string(), MfgMigrationTable { columns, rows });
    }
    let digest = migration_snapshot_digest(schema_version, &tables)?;
    Ok(MfgMigrationSnapshot {
        schema_version,
        tables,
        digest,
    })
}

fn import_migration_snapshot(
    connection: &mut Connection,
    snapshot: &MfgMigrationSnapshot,
) -> Result<(), MfgRepositoryError> {
    let expected_digest = migration_snapshot_digest(snapshot.schema_version, &snapshot.tables)?;
    if snapshot.digest != expected_digest {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "snapshot digest mismatch: expected {expected_digest}, received {}",
            snapshot.digest
        )));
    }
    let local_schema_version = schema_version(connection)?;
    if snapshot.schema_version != local_schema_version {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "schema version mismatch: snapshot {}, target {local_schema_version}",
            snapshot.schema_version
        )));
    }
    if snapshot.tables.len() != MFG_MIGRATION_TABLES.len() {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "snapshot contains {} tables; expected {}",
            snapshot.tables.len(),
            MFG_MIGRATION_TABLES.len()
        )));
    }
    for table_name in MFG_MIGRATION_TABLES {
        let snapshot_table = snapshot.tables.get(*table_name).ok_or_else(|| {
            MfgRepositoryError::MigrationRejected(format!(
                "snapshot is missing required table {table_name}"
            ))
        })?;
        let local_columns = migration_columns(connection, table_name)?;
        if snapshot_table.columns != local_columns {
            return Err(MfgRepositoryError::MigrationRejected(format!(
                "table {table_name} schema differs from the target"
            )));
        }
        if *table_name != "mfg_live_epoch" && count_table(connection, table_name)? != 0 {
            return Err(MfgRepositoryError::MigrationRejected(format!(
                "target table {table_name} is not empty"
            )));
        }
    }

    let target_is_sqlite = connection.is_sqlite();
    let transaction = connection.transaction_with_behavior(TransactionBehavior::Immediate)?;
    transaction.execute("DELETE FROM mfg_live_epoch", params![])?;
    for table_name in MFG_MIGRATION_TABLES {
        let table = &snapshot.tables[*table_name];
        let names = table
            .columns
            .iter()
            .map(|column| quote_migration_identifier(&column.name))
            .collect::<Result<Vec<_>, _>>()?
            .join(", ");
        let placeholders = (1..=table.columns.len())
            .map(|index| format!("?{index}"))
            .collect::<Vec<_>>()
            .join(", ");
        let sql = format!(
            "INSERT INTO {} ({names}) VALUES ({placeholders})",
            quote_migration_identifier(table_name)?
        );
        for row in &table.rows {
            if row.len() != table.columns.len() {
                return Err(MfgRepositoryError::MigrationRejected(format!(
                    "table {table_name} row width {} differs from column width {}",
                    row.len(),
                    table.columns.len()
                )));
            }
            let values = row
                .iter()
                .zip(&table.columns)
                .map(|(value, column)| migration_value_to_database(value, column.kind))
                .collect::<Vec<_>>();
            transaction.execute(&sql, values)?;
        }
    }
    let imported = export_migration_snapshot(&transaction)?;
    if imported.digest != snapshot.digest {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "target digest {} differs from source {}",
            imported.digest, snapshot.digest
        )));
    }
    if !target_is_sqlite {
        transaction.query_row(
            "SELECT setval(pg_get_serial_sequence('mfg_projection_event', 'event_id'), COALESCE(MAX(event_id), 1), MAX(event_id) IS NOT NULL) FROM mfg_projection_event",
            params![],
            |row| row.get::<_, i64>(0),
        )?;
    }
    transaction.commit()?;
    Ok(())
}

fn migration_columns(
    connection: &impl crate::database::ConnectionLike,
    table_name: &str,
) -> Result<Vec<MfgMigrationColumn>, MfgRepositoryError> {
    quote_migration_identifier(table_name)?;
    let mut statement = connection.prepare(&format!("PRAGMA table_info(\"{table_name}\")"))?;
    let columns = statement
        .query_map(params![], |row| {
            let name = row.get::<_, String>(1)?;
            let data_type = row.get::<_, String>(2)?;
            Ok(MfgMigrationColumn {
                name,
                kind: migration_column_kind(&data_type),
            })
        })?
        .collect::<crate::database::Result<Vec<_>>>()?;
    if columns.is_empty() {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "required table {table_name} has no columns"
        )));
    }
    Ok(columns)
}

fn migration_column_kind(data_type: &str) -> MfgMigrationColumnKind {
    let normalized = data_type.to_ascii_uppercase();
    if normalized.contains("INT") || normalized.contains("SERIAL") {
        MfgMigrationColumnKind::Integer
    } else if normalized.contains("REAL")
        || normalized.contains("FLOAT")
        || normalized.contains("DOUBLE")
        || normalized.contains("NUMERIC")
        || normalized.contains("DECIMAL")
    {
        MfgMigrationColumnKind::Real
    } else if normalized.contains("BOOL") {
        MfgMigrationColumnKind::Boolean
    } else if normalized.contains("BLOB") || normalized.contains("BYTEA") {
        MfgMigrationColumnKind::Bytes
    } else {
        MfgMigrationColumnKind::Text
    }
}

fn migration_value_from_database(value: &crate::database::Value) -> MfgMigrationValue {
    match value {
        crate::database::Value::NullText
        | crate::database::Value::NullInteger
        | crate::database::Value::NullReal
        | crate::database::Value::NullBool => MfgMigrationValue::Null,
        crate::database::Value::Integer(value) => MfgMigrationValue::Integer(*value),
        crate::database::Value::Real(value) => MfgMigrationValue::Real(*value),
        crate::database::Value::Text(value) => MfgMigrationValue::Text(value.clone()),
        crate::database::Value::Bool(value) => MfgMigrationValue::Boolean(*value),
        crate::database::Value::Bytes(value) => MfgMigrationValue::Bytes(value.clone()),
    }
}

fn migration_value_to_database(
    value: &MfgMigrationValue,
    kind: MfgMigrationColumnKind,
) -> crate::database::Value {
    match value {
        MfgMigrationValue::Null => match kind {
            MfgMigrationColumnKind::Integer => crate::database::Value::NullInteger,
            MfgMigrationColumnKind::Real => crate::database::Value::NullReal,
            MfgMigrationColumnKind::Boolean => crate::database::Value::NullBool,
            MfgMigrationColumnKind::Text | MfgMigrationColumnKind::Bytes => {
                crate::database::Value::NullText
            }
        },
        MfgMigrationValue::Integer(value) => crate::database::Value::Integer(*value),
        MfgMigrationValue::Real(value) => crate::database::Value::Real(*value),
        MfgMigrationValue::Text(value) => crate::database::Value::Text(value.clone()),
        MfgMigrationValue::Boolean(value) => crate::database::Value::Bool(*value),
        MfgMigrationValue::Bytes(value) => crate::database::Value::Bytes(value.clone()),
    }
}

fn quote_migration_identifier(identifier: &str) -> Result<String, MfgRepositoryError> {
    if identifier.is_empty()
        || !identifier
            .bytes()
            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'_')
    {
        return Err(MfgRepositoryError::MigrationRejected(format!(
            "unsafe relational identifier {identifier:?}"
        )));
    }
    Ok(format!("\"{identifier}\""))
}

fn migration_snapshot_digest(
    schema_version: i64,
    tables: &BTreeMap<String, MfgMigrationTable>,
) -> Result<String, MfgRepositoryError> {
    let encoded = serde_json::to_vec(&(schema_version, tables))?;
    Ok(format!("sha256:{:x}", Sha256::digest(encoded)))
}
